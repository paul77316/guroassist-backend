   124	                $errors['kind'][] = 'Unknown entry kind.';
   125	            }
   126	            if ($date > $today) {
   127	                $errors['followup_date'][] = 'The date cannot be in the future.';

   129	            if ($date < $i->intervention_date->format('Y-m-d')) {
   130	                $errors['followup_date'][] = 'The date cannot be before the intervention date.';
   131	            }
   132	
   133	            // Transition rules (one table in InterventionStatus).
   134	            if (! in_array($to, InterventionStatus::ALL, true)) {

   137	                $stays = $kind === 'reschedule'
   138	                    ? $from !== InterventionStatus::COMPLETED
   139	                    : in_array($from, [InterventionStatus::IN_PROGRESS, InterventionStatus::REQUIRES_FOLLOW_UP], true);

   141	                    $errors['to_status'][] = 'This intervention is ' . InterventionStatus::label($from) . ' and must move to a new status.';
   142	                }

   155	            }

   162	            }

   165	            }

   168	            }

   172	            }

   191	            ]);
