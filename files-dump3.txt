    42	        $mine = $ca && ($user->isAdmin() || ($user->teacher && $ca->teacher_id === $user->teacher->id));

     8	/**
     9	 * The "student still in your classes" rule, used by the due list for interventions and communications.
    10	 * It is the same rule InterventionListService applies and StudentPolicy::view uses for teachers: the student
    11	 * must be enrolled in one of the user's own classes. Archived classes are soft-deleted, so they never count.
    12	 * The query must have a student() relation. A user without a teacher profile (an admin) matches nothing.
    13	 */
    14	class TeacherStudentScope
    15	{
    16	    public function apply(Builder $query, User $user): Builder
    17	    {
    18	        $teacher = $user->teacher;
    19	        if (! $teacher) {
    20	            return $query->whereRaw('1 = 0');
    21	        }
    22	
    23	        $classIds = $teacher->classAssignments()->pluck('id');
    24	
    25	        return $query->whereHas('student', fn ($s) => $s->whereHas(
    26	            'enrollments',
    27	            fn ($e) => $e->whereIn('class_assignment_id', $classIds)
    28	        ));
    29	    }
